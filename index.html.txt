  <div class="modal" id="addModal">
    <h3 style="margin-top:0; font-size:1rem;">Add Custom RSS Source</h3>
    <div style="display:flex; gap:8px; flex-wrap:wrap;">
      <input type="text" id="newCategory" placeholder="Category Name (e.g. Crypto)">
      <input type="text" id="newUrl" placeholder="RSS Feed URL (https://...)">
      <button onclick="addSource()">Save Source</button>

  <script>
    // Live, high-reliability feeds from major outlets
    const defaultSources = [
      { category: "CNBC Business & Markets", url: "https://www.cnbc.com/id/100003114/device/rss/rss.html" },
      { category: "NYT Business & Tech", url: "https://rss.nytimes.com/services/xml/rss/nyt/Business.xml" },
      { category: "Wall Street Journal", url: "https://feeds.a.dj.com/rss/WSJcomUSBusiness.xml" },
      { category: "CNBC Economy & Jobs", url: "https://www.cnbc.com/id/20910258/device/rss/rss.html" },
      { category: "TechCrunch (Big Tech)", url: "https://techcrunch.com/feed/" },
      { category: "Yahoo Finance & Filings", url: "https://finance.yahoo.com/news/rssindex" }
    ];

    const seenItems = new Set();

    function getSources() {
      // Version v5 resets the local storage cache to pick up all 6 live feeds
      const saved = localStorage.getItem('deck_sources_v5');
      return saved ? JSON.parse(saved) : defaultSources;
    }

    function saveSources(sources) {
      localStorage.setItem('deck_sources_v5', JSON.stringify(sources));
    }

    function saveAlertKeywords() {
      localStorage.setItem('deck_alerts_v5', document.getElementById('alertKeywords').value);
    }

    function loadAlertKeywords() {
      const saved = localStorage.getItem('deck_alerts_v5');
      if (saved) document.getElementById('alertKeywords').value = saved;
    }

    function toggleModal(id) {
      const el = document.getElementById(id);
      el.style.display = el.style.display === 'block' ? 'none' : 'block';
    }

    function addSource() {
      const category = document.getElementById('newCategory').value.trim();
      const url = document.getElementById('newUrl').value.trim();
      if (!category || !url) return alert('Please enter both a category name and RSS URL.');

      const sources = getSources();
      sources.push({ category, url });
      saveSources(sources);

      document.getElementById('newCategory').value = '';
      document.getElementById('newUrl').value = '';
      toggleModal('addModal');
      fetchFeeds();
    }

    function removeSource(index) {
      const sources = getSources();
      sources.splice(index, 1);
      saveSources(sources);
      fetchFeeds();
    }

    function enableNotifications() {
      if ("Notification" in window) {
        Notification.requestPermission().then(permission => {
          if (permission === "granted") {
            alert("Desktop alerts enabled! You will be notified when tracked keywords appear.");
          }
        });
      } else {
        alert("This browser does not support web notifications.");
      }
    }

    function checkAlerts(items) {
      const alertInput = document.getElementById('alertKeywords').value;
      if (!alertInput || Notification.permission !== "granted") return;

      const keywords = alertInput.toLowerCase().split(',').map(k => k.trim()).filter(Boolean);
      if (keywords.length === 0) return;

      items.forEach(item => {
        const titleLower = item.title.toLowerCase();
        const itemKey = item.guid || item.link;

        if (!seenItems.has(itemKey)) {
          seenItems.add(itemKey);
          const matched = keywords.find(kw => titleLower.includes(kw));
          if (matched) {
            new Notification(`🚨 Priority Alert: ${matched.toUpperCase()}`, {
              body: item.title,
              icon: "https://fav.farm/📈"
            });
          }
        }
      });
    }

    async function loadFeed(url) {
      const apiPrefix = "https://api.rss2json.com/v1/api.json?rss_url=";
      try {
        const res = await fetch(apiPrefix + encodeURIComponent(url));
        const data = await res.json();
        if (data.status === 'ok' && data.items) {
          return data.items;
        }
        return [];
      } catch (e) {
        console.error("Failed to load feed", url, e);
        return [];
      }
    }

    async function fetchFeeds() {
      const grid = document.getElementById('deckGrid');
      grid.innerHTML = '<p style="color: var(--muted);">Updating live feeds...</p>';
      
      const sources = getSources();
      let html = '';

      for (let i = 0; i < sources.length; i++) {
        const source = sources[i];
        const items = await loadFeed(source.url);
        
        checkAlerts(items);

        html += `
          <div class="column">
            <div class="column-title">
              <span>${source.category}</span>
              <button class="remove-btn" onclick="removeSource(${i})" title="Remove Source">✕</button>
            </div>
            <div class="news-list">
        `;

        if (items.length === 0) {
          html += `<p style="font-size:0.85rem; color:var(--muted);">No items available or feed updating...</p>`;
        }

        items.forEach(item => {
          const pubDate = item.pubDate ? new Date(item.pubDate).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) : '';
          
          let domain = "news";
          try { domain = new URL(item.link).hostname.replace('www.', ''); } catch(e){}
          const iconUrl = `https://www.google.com/s2/favicons?domain=${domain}&sz=32`;

          html += `
            <div class="news-card">
              <a href="${item.link}" target="_blank" rel="noopener">${item.title}</a>
              <div class="meta">
                <span class="source-tag">
                  <img src="${iconUrl}" width="14" height="14" alt="logo" onerror="this.style.display='none'">
                  ${item.author || domain}
                </span>
                <span>${pubDate}</span>
              </div>
            </div>
          `;
        });

        html += `</div></div>`;
      }

      grid.innerHTML = html;
      document.getElementById('lastUpdated').innerText = 'Updated ' + new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      filterNews();
    }

    function filterNews() {
      const input = document.getElementById('searchInput').value.toLowerCase();
      document.querySelectorAll('.news-card').forEach(card => {
        card.style.display = card.innerText.toLowerCase().includes(input) ? 'block' : 'none';
      });
    }

    // Initialize
    loadAlertKeywords();
    fetchFeeds();

    // Auto-refresh every 5 minutes
    setInterval(fetchFeeds, 300000);
  </script>
